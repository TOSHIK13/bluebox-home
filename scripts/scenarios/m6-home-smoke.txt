# M6-H2: Blue Box Home (:app-home) - смоук главного экрана на эмуляторе.
# Предусловие: debug-APK поставлен вручную (`gradlew :app-home:assembleDebug`, затем adb -s emulator-NNNN install -r app-home/build/outputs/apk/debug/app-home-debug.apk);
# tv.ps1 install ставит только app-tv. Свои строки Watch Next кладёт debug-seed (4 шт., четвёртая на 98% и скрыта).
# Строки чужих приложений лаунчер без привилегий не видит - это проверяется отдельно, см. docs/plans/launcher.md.
adb "shell am force-stop app.bluebox.home.debug"
adb "shell am broadcast -a app.bluebox.home.SEED -n app.bluebox.home.debug/app.bluebox.home.SeedReceiver"
adb "shell am start -n app.bluebox.home.debug/app.bluebox.home.HomeActivity"
wait-for "Продолжить просмотр" 15
expect-text "VPN"
expect-text "Настройки"
expect-text "Дюна"
expect-text "Во все тяжкие"
expect-no-text "Начало"
expect-focus "Дюна"
# шестерёнка открывает панель быстрых настроек справа: «Все настройки», плитки, карточки VPN и разработчика
keys "UP"
keys "RIGHT*2"
expect-focus "Настройки"
keys "CENTER"
wait-for "Все настройки" 10
expect-text "Wi-Fi"
expect-text "Bluetooth"
expect-focus "Все настройки"
# «Все настройки» открывает настройки Home (переключатель фона и системные пункты)
keys "CENTER"
wait-for "Фон карточки" 10
expect-text "Wi-Fi"
expect-text "Bluetooth"
expect-text "VPN"
expect-text "Разрешения"
# Back возвращает в панель, ещё Back - на Home; фокус остаётся на шестерёнке, Home не закрывается
keys "BACK"
wait-for "Все настройки" 10
keys "BACK"
wait-for "Продолжить просмотр" 10
keys "BACK"
expect-text "Продолжить просмотр"
expect-focus "Настройки"
# вниз в док до «Все приложения»
keys "DOWN*3"
expect-focus "Все приложения"
keys "CENTER"
wait 2
# Back возвращает на Home, фокус возвращается на плитку «Все приложения»
keys "BACK"
wait-for "Продолжить просмотр" 10
expect-focus "Все приложения"
# долгое OK на плитке панели открывает редактор: «Добавить» и «Недавнее» (по умолчанию 5 плиток «недавних»)
adb "shell input keyevent --longpress KEYCODE_DPAD_CENTER"
wait-for "Панель приложений" 10
expect-text "Добавить"
expect-text "Недавнее"
# «Все приложения» - не слот: меню значка не показывается, фокус на «Добавить»
expect-focus "Добавить"
keys "BACK"
wait-for "Продолжить просмотр" 10
expect-focus "Все приложения"
